# %%
from pathlib import Path


def ler_frases(caminho):
    linhas = Path(caminho).read_text(encoding="utf-8-sig").splitlines()

    frases = []

    for linha in linhas:
        frase = linha.strip()

        if frase:
            frases.append(frase)

    return frases


frases = ler_frases("frases.txt")
print(f"{len(frases)} frases carregadas de frases.txt")

# %% [markdown]
# ## 2. Paráfrase
#
# Gera a paráfrase por retrotradução com dois modelos pequenos do Helsinki-NLP (MarianMT), rodando no próprio notebook: traduz a frase do português para o inglês e depois de volta para o português.

# %%
import torch
from transformers import MarianMTModel, MarianTokenizer

dispositivo = "cuda" if torch.cuda.is_available() else "cpu"
print(f"Dispositivo: {dispositivo}")

tokenizador_pt_en = MarianTokenizer.from_pretrained("Helsinki-NLP/opus-mt-ROMANCE-en")
modelo_pt_en = MarianMTModel.from_pretrained("Helsinki-NLP/opus-mt-ROMANCE-en").to(dispositivo)

tokenizador_en_pt = MarianTokenizer.from_pretrained("Helsinki-NLP/opus-mt-en-ROMANCE")
modelo_en_pt = MarianMTModel.from_pretrained("Helsinki-NLP/opus-mt-en-ROMANCE").to(dispositivo)


def traduzir(frase, tokenizador, modelo):
    entrada = tokenizador(frase, return_tensors="pt").to(dispositivo)
    saida = modelo.generate(**entrada)

    return tokenizador.decode(saida[0], skip_special_tokens=True)


def retraduzir(frase):
    ingles = traduzir(frase, tokenizador_pt_en, modelo_pt_en)
    portugues = traduzir(f">>pt_br<< {ingles}", tokenizador_en_pt, modelo_en_pt)

    return portugues


print(frases[0])
print(retraduzir(frases[0]))

# %% [markdown]
# Se a retrotradução falhar ou devolver a mesma frase, a frase original é mantida e o motivo fica registrado.

# %%
def gerar_parafrase(frase):
    try:
        parafrase = retraduzir(frase)
    except Exception as erro:
        return frase, f"original mantida: erro na tradução ({erro})"

    if parafrase.strip().casefold() == frase.strip().casefold():
        return frase, "original mantida: tradução repetiu a frase"

    return parafrase, "retrotradução pt → en → pt"


for frase in frases[:3]:
    parafrase, metodo = gerar_parafrase(frase)

    print(frase)
    print(parafrase)
    print(metodo)
    print()
