from pathlib import Path


def ler_frases(caminho):
    linhas = Path(caminho).read_text(encoding="utf-8-sig").splitlines()

    frases = []

    for linha in linhas:
        frase = linha.strip()

        if frase:
            frases.append(frase)

    return frases


frases = ler_frases("frases.txt")
print(f"{len(frases)} frases carregadas de frases.txt")

# %% [markdown]
# ## 2. Paráfrase
#
# Gera a paráfrase por retrotradução: traduz a frase do português para o inglês e depois de volta para o português. Todas as frases são traduzidas de uma vez em cada modelo. Se a retrotradução falhar ou devolver a mesma frase, a frase original é mantida e o motivo fica registrado.

# %%
def traduzir(textos, tokenizador, modelo):
    entrada = tokenizador(textos, return_tensors="pt", padding=True).to(dispositivo)
    saida = modelo.generate(**entrada)

    return tokenizador.batch_decode(saida, skip_special_tokens=True)


def retraduzir(frases):
    ingles = traduzir(frases, tokenizador_pt_en, modelo_pt_en)

    marcadas = []

    for texto in ingles:
        marcadas.append(f">>pt_br<< {texto}")

    return traduzir(marcadas, tokenizador_en_pt, modelo_en_pt)


def gerar_parafrases(frases):
    try:
        retraduzidas = retraduzir(frases)
    except Exception as erro:
        resultados = []

        for frase in frases:
            resultados.append((frase, f"original mantida: erro na tradução ({erro})"))

        return resultados

    resultados = []

    for frase, parafrase in zip(frases, retraduzidas):
        if parafrase.strip().casefold() == frase.strip().casefold():
            resultados.append((frase, "original mantida: tradução repetiu a frase"))
        else:
            resultados.append((parafrase, "retrotradução pt → en → pt"))

    return resultados


parafrases = gerar_parafrases(frases)
print(f"{len(parafrases)} paráfrases geradas")

# %% [markdown]
# Regra do número: troca o primeiro número da frase por outro valor. Números em algarismos são dobrados (anos ganham mais um), e números por extenso trocam pelo dobro.

# %%
def trocar_numero(texto):
    dobros = {
        "um": "dois",
        "dois": "quatro",
        "três": "seis",
        "quatro": "oito",
        "cinco": "dez",
        "seis": "doze",
        "sete": "catorze",
        "oito": "dezesseis",
        "nove": "dezoito",
        "dez": "vinte",
        "cem": "duzentos",
        "mil": "dois mil",
    }

    if texto.casefold() in dobros:
        return dobros[texto.casefold()]

    valor = float(texto.replace(".", "").replace(",", "."))

    if valor.is_integer() and 1900 <= valor <= 2100:
        return str(int(valor) + 1)

    novo = valor * 2

    if novo.is_integer():
        return str(int(novo))

    return str(novo).replace(".", ",")


def regra_numero(analise):
    for token in analise:
        if token.pos_ != "NUM":
            continue

        try:
            novo = trocar_numero(token.text)
        except ValueError:
            continue

        partes = []

        for outro in analise:
            if outro.i == token.i:
                partes.append(novo + outro.whitespace_)
            else:
                partes.append(outro.text_with_ws)

        return "".join(partes)

    return None


for analise in analises:
    contradicao = regra_numero(analise)

    if contradicao:
        print(analise.text)
        print(contradicao)
        print()

# %% [markdown]
# Regra da entidade: troca a primeira pessoa, lugar ou organização da frase por outra do mesmo tipo, tirada de uma lista fixa.

# %%
def regra_entidade(analise):
    substitutas = {
        "PER": ["Maria", "Pedro", "Ana", "João", "Carla"],
        "LOC": ["Curitiba", "Recife", "Porto Alegre", "Manaus", "Salvador"],
        "ORG": ["Petrobras", "Banco Central", "Vale", "Embraer", "Itaú"],
    }

    for entidade in analise.ents:
        if entidade.label_ not in substitutas:
            continue

        opcoes = substitutas[entidade.label_]

        if entidade.text in opcoes:
            posicao = opcoes.index(entidade.text)
            nova = opcoes[(posicao + 1) % len(opcoes)]
        else:
            nova = opcoes[0]

        texto = analise.text

        return texto[:entidade.start_char] + nova + texto[entidade.end_char:]

    return None


for analise in analises:
    contradicao = regra_entidade(analise)

    if contradicao:
        print(analise.text)
        print(contradicao)
        print()

# %% [markdown]
# Regra do antônimo: troca a primeira palavra encontrada num dicionário de opostos. Procura primeiro adjetivos, depois verbos e por último advérbios. Adjetivos mantêm gênero e número da palavra original.

# %%
def montar_opostos(pares):
    opostos = {}

    for palavra, oposta in pares:
        opostos[palavra] = oposta
        opostos[oposta] = palavra

    return opostos

def flexionar_adjetivo(palavra, token):
    if "Fem" in token.morph.get("Gender") and palavra.endswith("o"):
        palavra = palavra[:-1] + "a"

    if "Plur" in token.morph.get("Number"):
        if palavra.endswith("il"):
            palavra = palavra[:-2] + "eis"
        elif palavra.endswith("m"):
            palavra = palavra[:-1] + "ns"
        elif palavra.endswith(("r", "z", "s")):
            palavra = palavra + "es"
        elif palavra.endswith("l"):
            palavra = palavra[:-1] + "is"
        else:
            palavra = palavra + "s"

    return palavra


def trocar_token(analise, token, novo):
    if token.text[0].isupper():
        novo = novo[0].upper() + novo[1:]

    partes = []

    for outro in analise:
        if outro.i == token.i:
            partes.append(novo + outro.whitespace_)
        else:
            partes.append(outro.text_with_ws)

    return "".join(partes)


def verbo_negado(token):
    for filho in token.children:
        if filho.lower_ == "não":
            return True

    return False


def regra_antonimo(analise):
    for classe in ["ADJ", "VERB", "ADV"]:
        for token in analise:
            if token.pos_ != classe:
                continue

            if classe == "VERB" and verbo_negado(token):
                continue

            lema = token.lemma_.casefold()
            palavra = token.text.casefold()

            if classe == "ADJ" and lema in adjetivos_opostos:
                return trocar_token(analise, token, flexionar_adjetivo(adjetivos_opostos[lema], token))

            if classe == "VERB" and palavra in verbos_opostos:
                return trocar_token(analise, token, verbos_opostos[palavra])

            if classe == "ADV" and lema in adverbios_opostos:
                return trocar_token(analise, token, adverbios_opostos[lema])

    return None


for analise in analises:
    contradicao = regra_antonimo(analise)

    if contradicao:
        print(analise.text)
        print(contradicao)
        print()

# %% [markdown]
# Regra da negação: coloca "não" antes do verbo principal ou, se ele já estiver negado, tira o "não". Em frases como "O filme foi bom", o "não" vai antes do verbo de ligação ("não foi"). Se a frase não tiver verbo, ganha "Não é verdade que" no começo. Esta regra sempre funciona e é a última da fila.

# %%
def achar_verbo(analise):
    for token in analise:
        if token.dep_ == "ROOT":
            raiz = token

    auxiliares = []

    for filho in raiz.children:
        if filho.dep_ in {"aux", "aux:pass", "cop"} and filho.i < raiz.i:
            auxiliares.append(filho)

    if auxiliares:
        return raiz, auxiliares[0]

    if raiz.pos_ in {"VERB", "AUX"}:
        return raiz, raiz

    return raiz, None


def regra_negacao(analise):
    raiz, verbo = achar_verbo(analise)

    for filho in raiz.children:
        if filho.lower_ == "não":
            partes = []

            for token in analise:
                if token.i == filho.i:
                    continue

                if token.i == filho.i + 1 and filho.i == 0:
                    partes.append(token.text[0].upper() + token.text[1:] + token.whitespace_)
                else:
                    partes.append(token.text_with_ws)

            return "".join(partes)

    if verbo is None:
        texto = analise.text

        if analise[0].pos_ != "PROPN":
            texto = texto[0].lower() + texto[1:]

        return "Não é verdade que " + texto

    partes = []

    for token in analise:
        if token.i == verbo.i:
            if token.i == 0:
                partes.append("Não " + token.text.lower() + token.whitespace_)
            else:
                partes.append("não " + token.text_with_ws)
        else:
            partes.append(token.text_with_ws)

    return "".join(partes)


for analise in analises:
    print(analise.text)
    print(regra_negacao(analise))
    print()
